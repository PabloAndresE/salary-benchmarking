\section{El problema}
% ############################################################################

\begin{frame}{El problema es el denominador}
\begin{itemize}
\item Un benchmark salarial estima cuantiles de la distribución de sueldos
condicionada al puesto.
\item El estimador es una mediana ponderada. La especificación del condicionante
es el problema abierto.
\item La práctica establecida impone una taxonomía cerrada y traslada el mapeo al
cliente: manual, no reproducible entre codificadores, sin tasa de acuerdo
publicada.
\item Aquí el condicionante es la etiqueta declarada por el empleador, en texto
libre.
\end{itemize}
\end{frame}

\begin{frame}{El catálogo está en régimen de área pequeña}
\begin{itemize}
\item 65.181 títulos distintos, con mediana de 2 personas y \alert{1 empresa}
por título.
\item El 78\,\% de los títulos tiene una sola empresa detrás y cubre el 21\,\%
de las personas.
\item El 1\,\% con 30 o más empresas cubre el 46\,\%.
\item La mayoría de las celdas no admite estimación directa: el problema es de
estimación en áreas pequeñas, no de clasificación.
\end{itemize}
\end{frame}

\begin{frame}{El error irreducible acota la ganancia}
Con $y = \log(\text{sueldo}/\text{SBU})$, dentro de cada celda de cargo $c$:
\begin{equation*}
y_{fi} = \mu_c + u_f + e_{fi}, \qquad
u_f \sim (0, \tau^2), \qquad e_{fi} \sim (0, \sigma^2).
\end{equation*}
\begin{itemize}
\item Sobre el ajuste, $\tau = 0{,}2917$ y $\sigma = 0{,}1423$: el efecto
empleador explica el \alert{81\,\%} de la varianza residual.
\item Contra ese suelo, el techo de mejora por reagrupar es del
\alert{12,1\,\%}, y del 13,2\,\% sin la censura del salario básico.
\item Es la referencia contra la que se reporta, no el cero.
\end{itemize}
\end{frame}

\section{El método}
% ############################################################################

\begin{frame}{Cuantil ponderado por inverso de varianza}
El voto de la empresa $f$ es la mediana de sus $n_f$ personas en la celda; la
referencia, el cuantil ponderado al 50\,\% de los votos, con
\begin{equation*}
w_f = \frac{1}{\tau_c^2 + \sigma_c^2/n_f}.
\end{equation*}
\begin{itemize}
\item Los regímenes límite son las dos convenciones del campo:
$n_f \to \infty$ da un voto por empresa; $\tau_c^2 \to 0$, un voto por persona.
\item El ratio de pesos lo acota la razón de varianzas,
$w_{\max}/w_{\min} \le 1 + \sigma_c^2/\tau_c^2 \approx \alert{1{,}24}$, no el
tamaño relativo de las empresas.
\item $\tau_c$ y $\sigma_c$ se estiman por celda: el estimador no tiene
hiperparámetros.
\end{itemize}
\end{frame}

\begin{frame}{La similitud no es el peso del vecino}
Ponderar por similitud $\times$ precisión invierte el orden correcto:
\begin{table}
\centering
\small
\begin{tabular}{lrr}
\toprule
Vecinos de \texttt{OPERADOR DE PARQUEADERO} & Similitud & Peso \\
\midrule
\texttt{OPERADOR DE PARQUEADEROS} & 0,989 & 8\,\% \\
\texttt{OPERADOR DE EXCAVADORA} & 0,825 & \alert{46\,\%} \\
\bottomrule
\end{tabular}
\end{table}
Cada vecino $j$ se trata como observación ruidosa del puesto objetivo:
\begin{equation*}
\mathrm{var}_j = 1/W_j + \lambda\,(1-s_j), \qquad w_j = 1/\mathrm{var}_j.
\end{equation*}
$\lambda$ se estima por escalón jerárquico, donde varía un factor 18. Si todos
los vecinos están lejos, el segundo término domina: el intervalo se ensancha y la
confianza cae, sin regla ad hoc.
\end{frame}

\begin{frame}{Consolidación de grafías: tres reglas separables}
\begin{enumerate}
\item \textbf{Semántica.} Enlace completo sobre el grafo de similitud, umbral
0,95. El enlace simple percola: una componente de \alert{1.758} títulos con
\texttt{ASISTENTE CONTABLE} y \texttt{AUXILIAR DE LIMPIEZA} dentro.
\item \textbf{Ortográfica.} Indexado por variantes de borrado y absorción
asimétrica: el grupo raro con $\le 2$ empresas entra en el común con $\ge 10$.
\item \textbf{Flexión de género}, conservando la diferencia medida antes de
unir.
\end{enumerate}
\medskip
Cobertura directa: 57,7\,\% a \alert{64,1\,\%}. El coseno no separa erratas de
una letra: $s(\texttt{VENDEROR}, \texttt{VENDEDOR}) = 0{,}7177$.
\end{frame}

\begin{frame}{Protocolo, fijado antes del modelo}
\begin{itemize}
\item Partición \textbf{por empresa}: con partición por fila la empresa de
evaluación aparece en ajuste y el acierto es espurio.
\item Remuestreo \textbf{por empresa} en todo intervalo.
\item Regla de puntuación propia: pérdida \textit{pinball} en
$q \in \{0{,}25;\, 0{,}75\}$ sobre la banda de empresas.
\item Criterio de adopción declarado antes de correr cada experimento.
\item \textbf{Placebo de permutación} en toda reagrupación: en D-031 permutar
los destinos cuesta $+0{,}00130$ frente a $-0{,}00001$ de la regla real; en
D-018 el placebo seleccionó 118 cargos contra 93 de la regla, y la descartó.
\item El 20\,\% de evaluación final permanece intacto.
\end{itemize}
\end{frame}

\section{Resultados medidos}
% ############################################################################

\begin{frame}{La hipótesis de partida se rechazó}
\begin{itemize}
\item Hipótesis: la composición del pago ---fijo, comisiones, extras---
discrimina puestos dentro de una etiqueta ancha.
\item Fuera de muestra, $R^2 = \alert{+0{,}0042}$ contra placebo $-0{,}0002$: un
0,21\,\% del techo del 12\,\%. Se descarta.
\item El eje con poder discriminante es el nivel jerárquico,
$\omega^2 = \alert{0{,}2462}$, con un recorrido de $2{,}11\times$ entre el
escalón más bajo y el más alto.
\item El protocolo se construyó antes que el modelo y su primer resultado fue
falsificar la premisa del proyecto.
\end{itemize}
\end{frame}

\begin{frame}{Hipótesis medidas y no adoptadas}
\begin{itemize}
\item Sector para estrechar la banda: pinball 0,1289 contra 0,1255 sin
segmentar (D-022).
\item Sector para corregir el centro: no se sostiene fuera de muestra (D-022).
\item Escalera estimada con medianas: la media domina, $+0{,}00069$ (D-024).
\item División o clase CIIU en lugar de sección: degrada con la finura; la
sección acierta un 9,2\,\% más (D-029).
\item Contexto en el embedding: mejora el vecindario, pero el efecto sobre las
bandas cruza cero (D-030).
\end{itemize}
\end{frame}

\begin{frame}{Sector y tamaño: la misma pregunta, respuesta opuesta}
\begin{itemize}
\item \textbf{Sector.} Adoptado en D-014 y retirado en D-022. De la varianza de
pago \textit{entre} empresas para un mismo cargo, la división CIIU explica
\alert{6 puntos netos}: 0,389 en crudo contra 0,325 permutando. El resto es
inflación mecánica por partición en grupos pequeños.
\item \textbf{Tamaño.} No estrecha la banda ---la incertidumbre del centro sube
más de lo que baja el ancho--- pero corrige un sesgo de nivel en los cargos
altos.
\end{itemize}
\end{frame}
